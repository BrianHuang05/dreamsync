## Goal

Implement `ShowCache` — a file-based key-value store for compiled `ShowTimeline`s, keyed by composite `(track_id, profile_fingerprint)`. One JSON file per compiled show, organized in per-track directories under `~/.dreamsync/cache/`.

## Data Classes

### CacheEntry

```python
@dataclass(frozen=True)
class CacheEntry:
    """Metadata about one cached show (for listing/display)."""
    track_id: str
    profile_fingerprint: str
    profile_name: str        # from timeline metadata._cache_profile_name
    track_name: str          # from timeline metadata.track_name
    artist: str              # from timeline metadata.artist
    duration: float          # song duration in seconds
    compiled_at: str         # ISO 8601 timestamp from metadata._cache_compiled_at
    file_path: Path          # absolute path to the .show.json
    file_size: int           # bytes on disk (os.path.getsize)
```

### CacheStats

```python
@dataclass(frozen=True)
class CacheStats:
    """Aggregate cache statistics."""
    entry_count: int         # total .show.json files
    track_count: int         # unique track directories
    total_bytes: int         # sum of all file sizes
    cache_dir: Path          # resolved cache directory
```

## On-Disk Layout

```
~/.dreamsync/cache/
├── 4iV5W6fMJg0KuzBp/                    # Spotify track ID directory
│   ├── a1b2c3d4.show.json               # profile fingerprint a1b2c3d4
│   └── e5f6g7h8.show.json               # different profile
├── my_song_9f3c2a1b/                     # path-based track ID (offline mp3)
│   └── 00000000.show.json               # None profile (built-in defaults)
```

- **Directory per track** — enables fast per-track operations (`invalidate`, listing).
- **Filename = `{profile_fingerprint}.show.json`** — 8 hex chars + extension.
- **File contents** — `ShowTimeline` JSON (same format as `to_json()`) with extra `_cache_*` fields injected into `metadata`.

## Method Implementations — Detailed

### `__init__(self, cache_dir)`

```python
def __init__(self, cache_dir: Path | str = "~/.dreamsync/cache") -> None:
    self._cache_dir = Path(cache_dir).expanduser().resolve()
    self._cache_dir.mkdir(parents=True, exist_ok=True)
```

- Expand `~`, resolve to absolute.
- Create directory tree if it doesn't exist.
- Store as private `_cache_dir`.

### `cache_dir` property

```python
@property
def cache_dir(self) -> Path:
    return self._cache_dir
```

### `get(self, track_id, profile)`

```python
def get(self, track_id, profile=None):
    fp = profile_fingerprint(profile)
    path = self._entry_path(track_id, fp)
    if not path.exists():
        return None
    return self._read_timeline(path)
```

**`_read_timeline` error handling**: On any exception (`json.JSONDecodeError`, `KeyError`, `TypeError`, `ValueError`, etc.):
1. Log a warning: `"Corrupt cache file, removing: {path}"`
2. Delete the corrupt file: `path.unlink(missing_ok=True)`
3. Return `None` (degrade to cache miss)

This handles: partial writes from crashes, user tampering, schema changes after upgrades.

### `put(self, track_id, timeline, profile)`

```python
def put(self, track_id, timeline, profile=None):
    fp = profile_fingerprint(profile)
    path = self._entry_path(track_id, fp)
    self._write_timeline(path, timeline, profile)
    return path
```

**`_write_timeline` implementation — atomic write**:

```python
def _write_timeline(self, path, timeline, profile):
    # 1. Build the dict and augment metadata
    data = timeline.to_dict()
    data["metadata"] = dict(data.get("metadata", {}))  # mutable copy
    data["metadata"]["_cache_profile_name"] = profile.name if profile else "(built-in defaults)"
    data["metadata"]["_cache_profile_fp"] = profile_fingerprint(profile)
    data["metadata"]["_cache_compiled_at"] = datetime.now(timezone.utc).isoformat()

    # 2. Ensure track directory exists
    path.parent.mkdir(parents=True, exist_ok=True)

    # 3. Write to temp file in the same directory (same filesystem for rename)
    tmp_path = path.with_suffix(".tmp")
    tmp_path.write_text(json.dumps(data, indent=2), encoding="utf-8")

    # 4. Atomic rename (os.replace is atomic on most filesystems)
    os.replace(tmp_path, path)
```

**Why `path.with_suffix(".tmp")` instead of `tempfile.NamedTemporaryFile`?**
- `NamedTemporaryFile` may create the temp in a different directory (different filesystem on Linux), making `os.replace` fail with cross-device link error.
- Using `.tmp` suffix in the same directory guarantees same-filesystem rename.
- The `.tmp` file is overwritten each time — safe for single-writer.

**Metadata augmentation detail**: `ShowTimeline` is frozen, so we can't modify its `metadata` dict. Instead, we operate on the mutable dict returned by `to_dict()`, inject the `_cache_*` keys, then write the JSON. The original `ShowTimeline` object is unchanged. When read back via `from_dict()`, the `_cache_*` fields are preserved in `metadata` (it's a plain dict, no schema validation on unknown keys).

### `has(self, track_id, profile)`

```python
def has(self, track_id, profile=None):
    fp = profile_fingerprint(profile)
    return self._entry_path(track_id, fp).exists()
```

Cheaper than `get()` — no JSON parsing, just a stat call.

### `invalidate(self, track_id)`

```python
def invalidate(self, track_id):
    track_dir = self._track_dir(track_id)
    if not track_dir.exists():
        return 0
    count = 0
    for entry in track_dir.iterdir():
        if entry.suffix == ".json" and entry.name.endswith(".show.json"):
            entry.unlink()
            count += 1
    # Remove the now-empty directory
    try:
        track_dir.rmdir()
    except OSError:
        pass  # directory not empty (unexpected files) — leave it
    return count
```

- Deletes all `.show.json` files for the given track (all profile variants).
- Tries to clean up the empty directory.
- Returns 0 without error for unknown track IDs.

### `clear(self)`

```python
def clear(self):
    count = 0
    if not self._cache_dir.exists():
        return 0
    for track_dir in self._cache_dir.iterdir():
        if track_dir.is_dir():
            for entry in track_dir.iterdir():
                if entry.name.endswith(".show.json"):
                    entry.unlink()
                    count += 1
            try:
                track_dir.rmdir()
            except OSError:
                pass
    return count
```

- Iterates all track directories, deletes `.show.json` files, removes empty dirs.
- Returns total count of deleted files.

### `list_entries(self)`

```python
def list_entries(self):
    entries = []
    if not self._cache_dir.exists():
        return entries
    for track_dir in self._cache_dir.iterdir():
        if not track_dir.is_dir():
            continue
        track_id = track_dir.name
        for show_file in track_dir.iterdir():
            if not show_file.name.endswith(".show.json"):
                continue
            fp = show_file.stem.replace(".show", "")  # "a1b2c3d4.show.json" → "a1b2c3d4"
            timeline = self._read_timeline(show_file)
            if timeline is None:
                continue  # skip corrupt files
            meta = timeline.metadata
            entries.append(CacheEntry(
                track_id=track_id,
                profile_fingerprint=fp,
                profile_name=meta.get("_cache_profile_name", "unknown"),
                track_name=meta.get("track_name", "unknown"),
                artist=meta.get("artist", "unknown"),
                duration=timeline.duration,
                compiled_at=meta.get("_cache_compiled_at", "unknown"),
                file_path=show_file.resolve(),
                file_size=show_file.stat().st_size,
            ))
    entries.sort(key=lambda e: e.track_name.lower())
    return entries
```

### `stats(self)`

```python
def stats(self):
    entries = self.list_entries()
    track_ids = {e.track_id for e in entries}
    total_bytes = sum(e.file_size for e in entries)
    return CacheStats(
        entry_count=len(entries),
        track_count=len(track_ids),
        total_bytes=total_bytes,
        cache_dir=self._cache_dir,
    )
```

### Internal helpers

```python
def _entry_path(self, track_id: str, fp: str) -> Path:
    return self._cache_dir / track_id / f"{fp}.show.json"

def _track_dir(self, track_id: str) -> Path:
    return self._cache_dir / track_id

def _read_timeline(self, path: Path) -> ShowTimeline | None:
    try:
        return ShowTimeline.from_json(path)
    except Exception as exc:
        logger.warning("Corrupt cache file, removing: %s (%s)", path, exc)
        path.unlink(missing_ok=True)
        return None
```
