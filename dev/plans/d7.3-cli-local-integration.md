## Goal

Wire `LocalShowSession` into the CLI so users can play local audio files with synchronized lighting via two entry points:

1. **`dreamsync play song.mp3 --config devices.yaml`** — Enhanced play subcommand that compiles on-the-fly when `--show` is omitted.
2. **`dreamsync session --local song.mp3 --config devices.yaml`** — Session subcommand with `--local` flag for local mode.

Both entry points use the same underlying `run_local_session()` function.

### 3. `src/dreamsync/cli.py` — Update `play` handler in `main()`

```python
if args.command == "play":
    if args.show is not None:
        # Existing behavior: play pre-compiled show
        from .show.runtime import run_show_playback
        summary = run_show_playback(
            mp3_path=args.mp3_path,
            show_path=args.show,
            config_path=args.config,
            sample_rate=args.sample_rate,
            audio_device=args.audio_device,
            fps=args.fps,
            brightness=args.brightness,
            mirror=args.mirror,
            debug=getattr(args, "debug", False),
        )
    else:
        # New: on-the-fly compilation via LocalShowSession
        from .local_session import run_local_session
        from .output.auto_detect import detect_all_devices, build_multi_adapter
        from .output.govee_lan import load_device_config

        configs = load_device_config(args.config)
        detected = detect_all_devices(configs)
        multi_adapter = build_multi_adapter(detected, brightness=args.brightness, mirror=args.mirror)

        profile = None
        if getattr(args, "profile", None):
            from .profile import load_profile
            profile = load_profile(args.profile)

        stop_event = threading.Event()
        import signal
        signal.signal(signal.SIGINT, lambda *_: stop_event.set())

        summary = run_local_session(
            multi_adapter,
            args.mp3_path,
            cache_dir=getattr(args, "cache_dir", "~/.dreamsync/cache"),
            profile=profile,
            sample_rate=args.sample_rate,
            audio_device=args.audio_device,
            stop_event=stop_event,
            debug=getattr(args, "debug", False),
        )
    print(json.dumps(summary, separators=(",", ":")))
    return 0
```

class TestCLIIntegration:
    """CLI arg parsing, validation, and session wiring."""

    def test_play_show_optional(self):
        """--show is optional on the play subcommand."""
        from dreamsync.cli import build_parser

        parser = build_parser()
        args = parser.parse_args([
            "play", "song.mp3", "--config", "devices.yaml",
        ])
        assert args.show is None
        assert args.mp3_path == Path("song.mp3")

    def test_play_show_provided(self):
        """--show can still be provided explicitly."""
        from dreamsync.cli import build_parser

        parser = build_parser()
        args = parser.parse_args([
            "play", "song.mp3", "--show", "show.json", "--config", "devices.yaml",
        ])
        assert args.show == Path("show.json")

    def test_session_local_flag_parsed(self):
        """--local flag is parsed with audio path."""
        from dreamsync.cli import build_parser

        parser = build_parser()
        args = parser.parse_args([
            "session", "--config", "devices.yaml", "--local", "song.mp3",
        ])
        assert args.local == "song.mp3"

    def test_session_local_default_none(self):
        """--local defaults to None when not specified."""
        from dreamsync.cli import build_parser

        parser = build_parser()
        args = parser.parse_args([
            "session", "--config", "devices.yaml",
        ])
        assert args.local is None

    def test_play_cache_dir_parsed(self):
        """--cache-dir is parsed on the play subcommand."""
        from dreamsync.cli import build_parser

        parser = build_parser()
        args = parser.parse_args([
            "play", "song.mp3", "--config", "devices.yaml",
            "--cache-dir", "/tmp/cache",
        ])
        assert args.cache_dir == "/tmp/cache"

## Usage Examples

```bash
# Single file with on-the-fly compilation (new)
dreamsync play song.mp3 --config devices.yaml

# Single file with specific profile (new)
dreamsync play song.mp3 --config devices.yaml --profile chill

# Single file with pre-compiled show (existing behavior, unchanged)
dreamsync play song.mp3 --show compiled.json --config devices.yaml

# Local mode via session subcommand (new)
dreamsync session --config devices.yaml --local song.mp3

# Local mode with profile and cache (new)
dreamsync session --config devices.yaml --local song.mp3 --profile hype --cache-dir /tmp/cache

# Error: mutually exclusive (new)
dreamsync session --config devices.yaml --local song.mp3 --spotify --v3
# → "Error: --local and --v3 are mutually exclusive."
```
